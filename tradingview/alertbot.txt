//@version=6
indicator("Momentum + RSI · Oportunidades v8", overlay=false, max_lines_count=400, max_labels_count=400)

//==============================================================================
// GRUPOS
//==============================================================================
string GRP_RSI   = "RSI"
string GRP_SQZ   = "Squeeze Momentum"
string GRP_DIV   = "Divergencias"
string GRP_CONF  = "Señales por confluencia"
string GRP_VIS   = "Visualización"
string GRP_ALERT = "Alertas"

//==============================================================================
// RSI
//==============================================================================
bool  showRsiSignals = input.bool(true, "Mostrar disparadores RSI en precio", group=GRP_RSI)
int   rsiLen         = input.int(14, "Período RSI", minval=1, group=GRP_RSI)
float buyLevel       = input.float(30.0, "Nivel de sobreventa", minval=1, maxval=50, group=GRP_RSI)
float sellLevel      = input.float(70.0, "Nivel de sobrecompra", minval=50, maxval=99, group=GRP_RSI)

string panelMode = input.string("Momentum", "Panel inferior", options=["Momentum", "RSI"], group=GRP_VIS)
bool rsiPanel = panelMode == "RSI"
float rsi = ta.rsi(close, rsiLen)
bool rsiCrossUp = ta.crossover(rsi, buyLevel)
bool rsiCrossDown = ta.crossunder(rsi, sellLevel)
bool rsiBullTrigger = barstate.isconfirmed and rsiCrossUp
bool rsiBearTrigger = barstate.isconfirmed and rsiCrossDown

//==============================================================================
// SQUEEZE MOMENTUM (base LazyBear)
//==============================================================================
int   bbLength  = input.int(20, "BB Length", minval=1, group=GRP_SQZ)
float bbMult    = input.float(2.0, "BB Mult", minval=0.1, step=0.1, group=GRP_SQZ)
int   kcLength  = input.int(20, "KC Length", minval=1, group=GRP_SQZ)
float kcMult    = input.float(1.5, "KC Mult", minval=0.1, step=0.1, group=GRP_SQZ)
bool  showSqzSignals = input.bool(true, "Mostrar señales SQZMOM en precio", group=GRP_SQZ)

float bbBasis = ta.sma(close, bbLength)
float bbDev   = bbMult * ta.stdev(close, bbLength)
float upperBB = bbBasis + bbDev
float lowerBB = bbBasis - bbDev

float kcBasis = ta.sma(close, kcLength)
float rangeMa = ta.sma(ta.tr, kcLength)
float upperKC = kcBasis + rangeMa * kcMult
float lowerKC = kcBasis - rangeMa * kcMult

bool sqzOn  = lowerBB > lowerKC and upperBB < upperKC
bool sqzOff = lowerBB < lowerKC and upperBB > upperKC

float highestKC = ta.highest(high, kcLength)
float lowestKC  = ta.lowest(low, kcLength)
float meanKC    = math.avg(math.avg(highestKC, lowestKC), ta.sma(close, kcLength))
float sqzMom    = ta.linreg(close - meanKC, kcLength, 0)

color histColor = sqzMom > 0 ? (sqzMom > sqzMom[1] ? color.lime : color.green) : (sqzMom < sqzMom[1] ? color.red : color.maroon)
color sqzColor  = sqzOn ? color.gray : sqzOff ? color.blue : color.black

plot(rsiPanel ? na : sqzMom, "Momentum", style=plot.style_histogram, linewidth=4, color=histColor)
plot(rsiPanel ? na : 0, "Estado Squeeze", style=plot.style_cross, linewidth=2, color=sqzColor)
plot(rsiPanel ? rsi : na, "RSI panel", color=color.purple, linewidth=2)
plot(rsiPanel ? buyLevel : na, "RSI banda inferior", color=color.new(color.green, 50))
plot(rsiPanel ? sellLevel : na, "RSI banda superior", color=color.new(color.red, 50))
hline(0, "Cero", color=color.new(color.white, 55), linestyle=hline.style_dotted)

// Giro de momentum: reversión posible mientras el histograma sigue del otro lado de cero.
bool sqzBullTurn = barstate.isconfirmed and sqzMom < 0 and sqzMom > sqzMom[1] and sqzMom[1] <= sqzMom[2]
bool sqzBearTurn = barstate.isconfirmed and sqzMom > 0 and sqzMom < sqzMom[1] and sqzMom[1] >= sqzMom[2]

// Salida del squeeze: posible continuación cuando se libera la compresión.
bool sqzRelease   = barstate.isconfirmed and sqzOn[1] and not sqzOn
bool sqzBreakBull = sqzRelease and sqzMom > 0 and sqzMom > sqzMom[1]
bool sqzBreakBear = sqzRelease and sqzMom < 0 and sqzMom < sqzMom[1]
bool sqzBullEvent = sqzBullTurn or sqzBreakBull
bool sqzBearEvent = sqzBearTurn or sqzBreakBear

//==============================================================================
// DIVERGENCIAS REGULARES CONFIRMADAS
//==============================================================================
bool   showMomDivs       = input.bool(true, "Dibujar divergencias SQZMOM", group=GRP_DIV)
bool   showRsiDivs       = input.bool(true, "Dibujar divergencias RSI en precio", group=GRP_DIV)
int    pivotLen          = input.int(5, "Velas para confirmar el pivote", minval=1, maxval=20, group=GRP_DIV)
int    divRangeMin       = input.int(5, "Mínimo de barras entre pivotes", minval=1, group=GRP_DIV)
int    divRangeMax       = input.int(60, "Máximo de barras entre pivotes", minval=2, group=GRP_DIV)
bool   showEarlyMomDivs  = input.bool(true, "Dibujar preaviso SQZMOM con giro de 1 vela", group=GRP_DIV,
     tooltip="Líneas discontinuas: candidatas que pueden desaparecer. PRE: primer giro al cierre en 4h y diario.")

float pricePivotHigh = ta.pivothigh(high, pivotLen, pivotLen)
float pricePivotLow  = ta.pivotlow(low, pivotLen, pivotLen)
float momPivotHigh   = ta.pivothigh(sqzMom, pivotLen, pivotLen)
float momPivotLow    = ta.pivotlow(sqzMom, pivotLen, pivotLen)

// SQZMOM usa pivotes del propio histograma para que las líneas unan extremos
// visuales reales. RSI usa pivotes del precio y compara el RSI en esas velas.
bool momHighFound = not na(momPivotHigh)
bool momLowFound  = not na(momPivotLow)
bool rsiHighFound = not na(pricePivotHigh)
bool rsiLowFound  = not na(pricePivotLow)

float currentMomHigh = momPivotHigh
float currentMomLow  = momPivotLow
float currentRsiHigh = rsi[pivotLen]
float currentRsiLow  = rsi[pivotLen]
float currentPriceHigh = high[pivotLen]
float currentPriceLow  = low[pivotLen]
int currentPivotBar = bar_index - pivotLen

var float lastMomHigh      = na
var float lastMomLow       = na
var float lastMomHighPrice = na
var float lastMomLowPrice  = na
var int   lastMomHighBar   = na
var int   lastMomLowBar    = na

var float lastRsiHigh      = na
var float lastRsiLow       = na
var float lastRsiHighPrice = na
var float lastRsiLowPrice  = na
var int   lastRsiHighBar   = na
var int   lastRsiLowBar    = na

bool momBearDiv = false
bool momBullDiv = false
bool rsiBearDiv = false
bool rsiBullDiv = false

// Referencias conocidas ANTES de este cierre, como en el scanner.
float earlyHigh = lastMomHigh
float earlyLow = lastMomLow
float earlyHighPrice = lastMomHighPrice
float earlyLowPrice = lastMomLowPrice
int earlyHighBar = lastMomHighBar
int earlyLowBar = lastMomLowBar

if barstate.isconfirmed and momHighFound
    int barsSince = na(lastMomHighBar) ? na : currentPivotBar - lastMomHighBar
    momBearDiv := not na(lastMomHigh) and lastMomHigh > 0 and currentMomHigh > 0 and currentPriceHigh > lastMomHighPrice and currentMomHigh < lastMomHigh and barsSince >= divRangeMin and barsSince <= divRangeMax
    if momBearDiv and showMomDivs and not rsiPanel
        line.new(lastMomHighBar, lastMomHigh, currentPivotBar, currentMomHigh, width=2, color=color.red)
    lastMomHigh      := currentMomHigh
    lastMomHighPrice := currentPriceHigh
    lastMomHighBar   := currentPivotBar

if barstate.isconfirmed and momLowFound
    int barsSince = na(lastMomLowBar) ? na : currentPivotBar - lastMomLowBar
    momBullDiv := not na(lastMomLow) and lastMomLow < 0 and currentMomLow < 0 and currentPriceLow < lastMomLowPrice and currentMomLow > lastMomLow and barsSince >= divRangeMin and barsSince <= divRangeMax
    if momBullDiv and showMomDivs and not rsiPanel
        line.new(lastMomLowBar, lastMomLow, currentPivotBar, currentMomLow, width=2, color=color.lime)
    lastMomLow      := currentMomLow
    lastMomLowPrice := currentPriceLow
    lastMomLowBar   := currentPivotBar

if barstate.isconfirmed and rsiHighFound
    int barsSince = na(lastRsiHighBar) ? na : currentPivotBar - lastRsiHighBar
    rsiBearDiv := not na(lastRsiHigh) and currentPriceHigh > lastRsiHighPrice and currentRsiHigh < lastRsiHigh and barsSince >= divRangeMin and barsSince <= divRangeMax
    if rsiBearDiv and showRsiDivs
        line.new(lastRsiHighBar, lastRsiHighPrice, currentPivotBar, currentPriceHigh, width=2, color=color.fuchsia, force_overlay=true)
    if rsiBearDiv and showRsiDivs
        label.new(bar_index, high, "RSI−", yloc=yloc.abovebar, style=label.style_label_down, color=color.new(color.fuchsia, 30), textcolor=color.white, size=size.tiny, force_overlay=true,
             tooltip="RSI: " + str.tostring(lastRsiHigh, "#.0") + " → " + str.tostring(currentRsiHigh, "#.0") + "\nPrecio: " + str.tostring(lastRsiHighPrice, format.mintick) + " → " + str.tostring(currentPriceHigh, format.mintick) + "\nConfirmada " + str.tostring(pivotLen) + " velas después del extremo.")
        if rsiPanel
            line.new(lastRsiHighBar, lastRsiHigh, currentPivotBar, currentRsiHigh, color=color.fuchsia, width=2)
    lastRsiHigh      := currentRsiHigh
    lastRsiHighPrice := currentPriceHigh
    lastRsiHighBar   := currentPivotBar

if barstate.isconfirmed and rsiLowFound
    int barsSince = na(lastRsiLowBar) ? na : currentPivotBar - lastRsiLowBar
    rsiBullDiv := not na(lastRsiLow) and currentPriceLow < lastRsiLowPrice and currentRsiLow > lastRsiLow and barsSince >= divRangeMin and barsSince <= divRangeMax
    if rsiBullDiv and showRsiDivs
        line.new(lastRsiLowBar, lastRsiLowPrice, currentPivotBar, currentPriceLow, width=2, color=color.aqua, force_overlay=true)
    if rsiBullDiv and showRsiDivs
        label.new(bar_index, low, "RSI+", yloc=yloc.belowbar, style=label.style_label_up, color=color.new(color.aqua, 30), textcolor=color.white, size=size.tiny, force_overlay=true,
             tooltip="RSI: " + str.tostring(lastRsiLow, "#.0") + " → " + str.tostring(currentRsiLow, "#.0") + "\nPrecio: " + str.tostring(lastRsiLowPrice, format.mintick) + " → " + str.tostring(currentPriceLow, format.mintick) + "\nConfirmada " + str.tostring(pivotLen) + " velas después del extremo.")
        if rsiPanel
            line.new(lastRsiLowBar, lastRsiLow, currentPivotBar, currentRsiLow, color=color.aqua, width=2)
    lastRsiLow      := currentRsiLow
    lastRsiLowPrice := currentPriceLow
    lastRsiLowBar   := currentPivotBar

//==============================================================================
// PREAVISO DE DIVERGENCIA SQZMOM (1 vela a la derecha)
//==============================================================================
float previousMomHigh = ta.highest(sqzMom[2], pivotLen)
float previousMomLow = ta.lowest(sqzMom[2], pivotLen)
int candidateBar = bar_index - 1
bool earlyReady = not na(sqzMom[pivotLen + 1]) and not na(sqzMom[1])
bool earlyTimeframe = timeframe.period == "240" or timeframe.period == "1D"
var int lastEarlyBearReference = na
var int lastEarlyBullReference = na
bool earlyBearDiv = barstate.isconfirmed and earlyReady and not na(earlyHighBar) and
     sqzMom[1] > 0 and earlyHigh > 0 and sqzMom[1] >= previousMomHigh and
     sqzMom[1] < earlyHigh and high[1] > earlyHighPrice and
     candidateBar - earlyHighBar >= divRangeMin and candidateBar - earlyHighBar <= divRangeMax and
     sqzMom < sqzMom[1] and (na(lastEarlyBearReference) or lastEarlyBearReference != earlyHighBar)
bool earlyBullDiv = barstate.isconfirmed and earlyReady and not na(earlyLowBar) and
     sqzMom[1] < 0 and earlyLow < 0 and sqzMom[1] <= previousMomLow and
     sqzMom[1] > earlyLow and low[1] < earlyLowPrice and
     candidateBar - earlyLowBar >= divRangeMin and candidateBar - earlyLowBar <= divRangeMax and
     sqzMom > sqzMom[1] and (na(lastEarlyBullReference) or lastEarlyBullReference != earlyLowBar)
if earlyBearDiv
    lastEarlyBearReference := earlyHighBar
if earlyBullDiv
    lastEarlyBullReference := earlyLowBar
bool earlyBearSetup = earlyBearDiv
bool earlyBullSetup = earlyBullDiv
earlyBearDiv := earlyBearDiv and earlyTimeframe
earlyBullDiv := earlyBullDiv and earlyTimeframe
if not rsiPanel and showMomDivs and showEarlyMomDivs and earlyBearDiv
    line.new(earlyHighBar, earlyHigh, candidateBar, sqzMom[1], color=color.orange, style=line.style_dashed)
    label.new(bar_index, sqzMom, "PRE Bear\n1 vela", style=label.style_label_down, color=color.orange, textcolor=color.white, size=size.tiny)
if not rsiPanel and showMomDivs and showEarlyMomDivs and earlyBullDiv
    line.new(earlyLowBar, earlyLow, candidateBar, sqzMom[1], color=color.aqua, style=line.style_dashed)
    label.new(bar_index, sqzMom, "PRE Bull\n1 vela", style=label.style_label_up, color=color.aqua, textcolor=color.black, size=size.tiny)

// Vista tentativa del extremo actual. Se reemplaza, no deja señales históricas falsas.
float formingHighLeft = ta.highest(sqzMom[1], pivotLen)
float formingLowLeft = ta.lowest(sqzMom[1], pivotLen)
bool formingBear = not na(lastMomHighBar) and sqzMom > 0 and lastMomHigh > 0 and
     high > lastMomHighPrice and sqzMom < lastMomHigh and sqzMom >= formingHighLeft and
     bar_index - lastMomHighBar >= divRangeMin and bar_index - lastMomHighBar <= divRangeMax and
     (na(lastEarlyBearReference) or lastEarlyBearReference != lastMomHighBar)
bool formingBull = not na(lastMomLowBar) and sqzMom < 0 and lastMomLow < 0 and
     low < lastMomLowPrice and sqzMom > lastMomLow and sqzMom <= formingLowLeft and
     bar_index - lastMomLowBar >= divRangeMin and bar_index - lastMomLowBar <= divRangeMax and
     (na(lastEarlyBullReference) or lastEarlyBullReference != lastMomLowBar)
var line formingLine = na
var label formingLabel = na
if barstate.islast
    line.delete(formingLine)
    label.delete(formingLabel)
    if not rsiPanel and showEarlyMomDivs and showMomDivs and earlyTimeframe and (formingBear or formingBull)
        formingLine := line.new(formingBear ? lastMomHighBar : lastMomLowBar, formingBear ? lastMomHigh : lastMomLow, bar_index, sqzMom, color=color.yellow, style=line.style_dashed)
        formingLabel := label.new(bar_index, sqzMom, "EN FORMACIÓN\nPuede invalidarse", style=label.style_label_left, color=color.yellow, textcolor=color.black, size=size.tiny)

//==============================================================================
// CONFLUENCIA
//==============================================================================
bool showConfluence = input.bool(true, "Mostrar BUY/SELL de oportunidad", group=GRP_CONF)
int setupWindow = input.int(3, "Vigencia del setup (velas)", minval=0, maxval=10, group=GRP_CONF)
int signalCooldown = input.int(4, "Espera entre señales iguales", minval=0, maxval=100, group=GRP_CONF)
bool showEma20 = input.bool(false, "Mostrar EMA20 (corto plazo)", group=GRP_VIS)
bool showEma50 = input.bool(true, "Mostrar EMA50", group=GRP_VIS)
bool showEma = input.bool(true, "Mostrar EMA200", group=GRP_VIS)
int emaLength = input.int(200, "EMA de contexto", minval=1, group=GRP_CONF)
float trendEma = ta.ema(close, emaLength)
float ema20 = ta.ema(close, 20)
float ema50 = ta.ema(close, 50)
bool bullTrendOk = close > trendEma
bool bearTrendOk = close < trendEma

f_recent(bool condition, int window) =>
    int age = ta.barssince(condition)
    not na(age) and age <= window

bool recentRsiBull = f_recent(rsiBullTrigger, setupWindow)
bool recentRsiBear = f_recent(rsiBearTrigger, setupWindow)
bool recentRsiBullDiv = f_recent(rsiBullDiv, setupWindow)
bool recentRsiBearDiv = f_recent(rsiBearDiv, setupWindow)
bool recentMomBullDiv = f_recent(momBullDiv, setupWindow)
bool recentMomBearDiv = f_recent(momBearDiv, setupWindow)
bool recentEarlyBull = f_recent(earlyBullSetup, setupWindow)
bool recentEarlyBear = f_recent(earlyBearSetup, setupWindow)
bool freshBullSetup = rsiBullTrigger or rsiBullDiv or momBullDiv or earlyBullSetup
bool freshBearSetup = rsiBearTrigger or rsiBearDiv or momBearDiv or earlyBearSetup
var int lastBullSetup = na
var int lastBearSetup = na
if freshBullSetup
    lastBullSetup := bar_index
if freshBearSetup
    lastBearSetup := bar_index
var int usedBullSetup = na
var int usedBearSetup = na
var int lastBuyBar = na
var int lastSellBar = na
bool bullReady = recentRsiBull or recentRsiBullDiv or recentMomBullDiv or recentEarlyBull
bool bearReady = recentRsiBear or recentRsiBearDiv or recentMomBearDiv or recentEarlyBear
bool buySignal = barstate.isconfirmed and bullReady and sqzMom > sqzMom[1] and close > high[1] and
     (na(usedBullSetup) or lastBullSetup > usedBullSetup) and (na(lastBuyBar) or bar_index - lastBuyBar > signalCooldown)
bool sellSignal = barstate.isconfirmed and bearReady and sqzMom < sqzMom[1] and close < low[1] and
     (na(usedBearSetup) or lastBearSetup > usedBearSetup) and (na(lastSellBar) or bar_index - lastSellBar > signalCooldown)
string bullReason = (recentRsiBull ? "RSI salió de sobreventa. " : "") + (recentRsiBullDiv ? "Divergencia RSI. " : "") + (recentMomBullDiv ? "Divergencia momentum confirmada. " : "") + (recentEarlyBull ? "Divergencia momentum tentativa. " : "")
string bearReason = (recentRsiBear ? "RSI salió de sobrecompra. " : "") + (recentRsiBearDiv ? "Divergencia RSI. " : "") + (recentMomBearDiv ? "Divergencia momentum confirmada. " : "") + (recentEarlyBear ? "Divergencia momentum tentativa. " : "")
if buySignal
    lastBuyBar := bar_index
    usedBullSetup := lastBullSetup
if sellSignal
    lastSellBar := bar_index
    usedBearSetup := lastBearSetup

//==============================================================================
// VISUALIZACIÓN
//==============================================================================
bool showConfirmedMarkers = input.bool(false, "Marcar confirmación de divergencias", group=GRP_VIS)

plot(showEma20 ? ema20 : na, "EMA20", color=color.new(color.gray, 30), linewidth=1, force_overlay=true)
plot(showEma50 ? ema50 : na, "EMA50", color=color.teal, linewidth=2, force_overlay=true)
plot(showEma ? trendEma : na, "EMA tendencia", color=color.orange, linewidth=2, force_overlay=true)

plotshape(showRsiSignals and rsiBullTrigger, "RSI sale de sobreventa", shape.circle, location.belowbar,
     color=color.new(color.lime, 15), size=size.tiny, display=display.pane, textcolor=color.white, force_overlay=true)
plotshape(showRsiSignals and rsiBearTrigger, "RSI sale de sobrecompra", shape.circle, location.abovebar,
     color=color.new(color.red, 15), size=size.tiny, display=display.pane, textcolor=color.white, force_overlay=true)

plotshape(showSqzSignals and sqzBullTurn, "Giro SQZ alcista", shape.diamond, location.belowbar,
     color=color.lime, size=size.tiny, display=display.pane, textcolor=color.black, force_overlay=true)
plotshape(showSqzSignals and sqzBearTurn, "Giro SQZ bajista", shape.diamond, location.abovebar,
     color=color.red, size=size.tiny, display=display.pane, textcolor=color.black, force_overlay=true)
plotshape(showSqzSignals and sqzBreakBull, "Salida squeeze alcista", shape.square, location.belowbar,
     color=color.lime, size=size.tiny, display=display.pane, textcolor=color.white, force_overlay=true)
plotshape(showSqzSignals and sqzBreakBear, "Salida squeeze bajista", shape.square, location.abovebar,
     color=color.red, size=size.tiny, display=display.pane, textcolor=color.white, force_overlay=true)

plotshape(showConfirmedMarkers and showMomDivs and momBullDiv, "SQZ Bull Div confirmada", shape.triangleup, location.bottom,
     color=color.lime, size=size.tiny, display=display.pane, text="M DIV+", textcolor=color.black)
plotshape(showConfirmedMarkers and showMomDivs and momBearDiv, "SQZ Bear Div confirmada", shape.triangledown, location.top,
     color=color.red, size=size.tiny, display=display.pane, text="M DIV-", textcolor=color.white)

if showConfluence and buySignal
    label.new(bar_index, low, "BUY", yloc=yloc.belowbar, style=label.style_label_up, color=color.green, textcolor=color.white, size=size.small, force_overlay=true,
         tooltip=bullReason + "Momentum sube y cierre supera el máximo anterior. " + (bullTrendOk ? "Encima de EMA200." : "Contra EMA200: posible rebote o giro."))
if showConfluence and sellSignal
    label.new(bar_index, high, "SELL", yloc=yloc.abovebar, style=label.style_label_down, color=color.red, textcolor=color.white, size=size.small, force_overlay=true,
         tooltip=bearReason + "Momentum cae y cierre pierde el mínimo anterior. " + (bearTrendOk ? "Debajo de EMA200." : "Contra EMA200: posible corrección o giro."))

plot(rsi, "RSI", display=display.data_window)
bool showContext = input.bool(true, "Resumen RSI y tendencia", group=GRP_VIS)
var table context = table.new(position.top_right, 2, 2, bgcolor=color.new(color.black, 20))
if barstate.islast and showContext
    table.cell(context, 0, 0, "RSI " + str.tostring(rsiLen), text_color=color.white)
    table.cell(context, 1, 0, str.tostring(rsi, "#.0"), text_color=rsi > sellLevel ? color.red : rsi < buyLevel ? color.lime : color.white)
    table.cell(context, 0, 1, "EMA " + str.tostring(emaLength), text_color=color.white)
    table.cell(context, 1, 1, close > trendEma ? "Precio encima" : "Precio debajo", text_color=close > trendEma ? color.lime : color.orange)

//==============================================================================
// ALERTAS
//==============================================================================
bool alertIndividual = input.bool(false, "Alertar señales individuales", group=GRP_ALERT)
bool alertDivs       = input.bool(false, "Alertar divergencias confirmadas", group=GRP_ALERT)
bool alertEarly      = input.bool(true, "Alertar preavisos SQZMOM", group=GRP_ALERT)
bool alertConfluence = input.bool(true, "Alertar BUY/SELL de oportunidad", group=GRP_ALERT)

string signals = ""

if alertConfluence and buySignal
    signals += "BUY de oportunidad: " + bullReason + "Cierre supera el máximo previo y momentum sube.\n"
if alertConfluence and sellSignal
    signals += "SELL de oportunidad: " + bearReason + "Cierre pierde el mínimo previo y momentum cae.\n"
if alertDivs and rsiBullDiv
    signals += "Divergencia RSI alcista confirmada con " + str.tostring(pivotLen) + " velas de demora.\n"
if alertDivs and rsiBearDiv
    signals += "Divergencia RSI bajista confirmada con " + str.tostring(pivotLen) + " velas de demora.\n"
if alertDivs and momBullDiv
    signals += "Divergencia SQZMOM alcista confirmada con " + str.tostring(pivotLen) + " velas de demora.\n"
if alertDivs and momBearDiv
    signals += "Divergencia SQZMOM bajista confirmada con " + str.tostring(pivotLen) + " velas de demora.\n"
if alertEarly and earlyBullDiv
    signals += "PRE divergencia SQZMOM alcista: giro de una vela; puede fallar.\n"
if alertEarly and earlyBearDiv
    signals += "PRE divergencia SQZMOM bajista: giro de una vela; puede fallar.\n"
if alertIndividual and rsiBullTrigger
    signals += "RSI salió de sobreventa.\n"
if alertIndividual and rsiBearTrigger
    signals += "RSI salió de sobrecompra.\n"
if alertIndividual and sqzBullEvent
    signals += "Evento SQZMOM alcista.\n"
if alertIndividual and sqzBearEvent
    signals += "Evento SQZMOM bajista.\n"

if barstate.isconfirmed and str.length(signals) > 0
    alert(syminfo.tickerid + " / " + timeframe.period + "\n" + signals, alert.freq_once_per_bar_close)

alertcondition(alertConfluence and buySignal, "BUY de oportunidad", "BUY de oportunidad confirmado al cierre")
alertcondition(alertConfluence and sellSignal, "SELL de oportunidad", "SELL de oportunidad confirmado al cierre")
alertcondition(alertDivs and rsiBullDiv, "RSI Bull Div", "Divergencia RSI alcista confirmada")
alertcondition(alertDivs and rsiBearDiv, "RSI Bear Div", "Divergencia RSI bajista confirmada")
alertcondition(alertDivs and momBullDiv, "SQZ Bull Div", "Divergencia SQZMOM alcista confirmada")
