//@version=6
indicator("SQZMOM + RSI Signal (precio) + Momentum Div (panel) v6 preaviso con giro de 1 vela", overlay=false, max_lines_count=300, max_labels_count=300)

//==============================================================================
// GRUPOS DE CONFIGURACIÓN
//==============================================================================
GRP_RSI   = "Configuración RSI"
GRP_SQZ   = "Configuración Squeeze Momentum"
GRP_DIV   = "Divergencia de Momentum"
GRP_VIS   = "Filtros Visuales"

//------------------------------------------------------------------------------
// RSI
//------------------------------------------------------------------------------
showRsiSignals = input.bool(true, "Mostrar Señales RSI en Precio", group=GRP_RSI)
rsiLen         = input.int(14, "Período RSI", minval=1, group=GRP_RSI)
buyLevel       = input.int(30, "Nivel Sobreventa (BUY)", minval=1, maxval=50, group=GRP_RSI)
sellLevel      = input.int(70, "Nivel Sobrecompra (SELL)", minval=50, maxval=99, group=GRP_RSI)

rsi = ta.rsi(close, rsiLen)

buyCross = ta.crossover(rsi, buyLevel)
buySignal = barstate.isconfirmed and buyCross
sellCross = ta.crossunder(rsi, sellLevel)
sellSignal = barstate.isconfirmed and sellCross

//------------------------------------------------------------------------------
// SQZMOM
//------------------------------------------------------------------------------
length    = input.int(20, "BB Length", minval=1, group=GRP_SQZ)
mult      = input.float(2.0, "BB Mult", minval=0.1, group=GRP_SQZ)
lengthKC  = input.int(20, "KC Length", minval=1, group=GRP_SQZ)
multKC    = input.float(1.5, "KC Mult", minval=0.1, group=GRP_SQZ)

basis   = ta.sma(close, length)
dev     = mult * ta.stdev(close, length)
upperBB = basis + dev
lowerBB = basis - dev

ma      = ta.sma(close, lengthKC)
rangema = ta.sma(ta.tr, lengthKC)
upperKC = ma + rangema * multKC
lowerKC = ma - rangema * multKC

sqzOn  = lowerBB > lowerKC and upperBB < upperKC
sqzOff = lowerBB < lowerKC and upperBB > upperKC

val = ta.linreg(close - math.avg(math.avg(ta.highest(high, lengthKC), ta.lowest(low, lengthKC)), ta.sma(close, lengthKC)), lengthKC, 0)

histColor = val > 0 ? (val > val[1] ? color.lime : color.green) : (val < val[1] ? color.red : color.maroon)
plot(val, title="Histograma Momentum", style=plot.style_histogram, linewidth=4, color=histColor)

// Puntos Squeeze
sqzColor = sqzOn ? color.gray : sqzOff ? color.blue : color.black
plot(0, title="Punto Squeeze", style=plot.style_cross, linewidth=2, color=sqzColor)
hline(0, color=color.new(color.white, 50), linestyle=hline.style_dotted)

//------------------------------------------------------------------------------
// PLOTSHAPE DE PRECIO (FORCE OVERLAY)
//------------------------------------------------------------------------------
plotshape(showRsiSignals and buySignal, title="BUY Signal", style=shape.triangleup, 
          location=location.belowbar, color=color.green, size=size.small, 
          text="BUY", textcolor=color.white, force_overlay=true)

plotshape(showRsiSignals and sellSignal, title="SELL Signal", style=shape.triangledown, 
          location=location.abovebar, color=color.red, size=size.small, 
          text="SELL", textcolor=color.white, force_overlay=true)

//------------------------------------------------------------------------------
// DIVERGENCIAS DE MOMENTUM
//------------------------------------------------------------------------------
showDivs    = input.bool(true, "Dibujar Divergencias", group=GRP_DIV)
pivotLen    = input.int(5, "Pivot Length", minval=1, group=GRP_DIV, tooltip="Barras a izquierda/derecha para confirmar pivote")
divRangeMin = input.int(5, "Mínimo de Barras entre Pivotes", minval=1, group=GRP_DIV)
divRangeMax = input.int(60, "Máximo de Barras entre Pivotes", minval=1, group=GRP_DIV)

showConfirmed = input.bool(false, "Mostrar divergencias confirmadas", group=GRP_DIV)

momHigh = ta.pivothigh(val, pivotLen, pivotLen)
momLow  = ta.pivotlow(val, pivotLen, pivotLen)

var float lastHigh      = na
var float lastLow       = na
var float lastHighPrice = na
var float lastLowPrice  = na
var int   lastHighBar   = na
var int   lastLowBar    = na

momBearDiv = false
momBullDiv = false

// Extremo candidato en la vela ANTERIOR, con cinco barras a izquierda.
// La vela actual debe cerrar con giro estricto de momentum: una barra a derecha.
// La referencia sigue siendo el ultimo pivot confirmado con pivotLen barras a derecha.
previousHigh = ta.highest(val[2], pivotLen)
previousLow = ta.lowest(val[2], pivotLen)
candidateBar = bar_index - 1
ready = not na(val[pivotLen + 1]) and not na(val[1])
earlyBearStart = barstate.isconfirmed and ready and not na(lastHighBar) and val[1] >= previousHigh and val[1] < lastHigh and high[1] > lastHighPrice and candidateBar - lastHighBar >= divRangeMin and candidateBar - lastHighBar <= divRangeMax and val < val[1]
earlyBullStart = barstate.isconfirmed and ready and not na(lastLowBar) and val[1] <= previousLow and val[1] > lastLow and low[1] < lastLowPrice and candidateBar - lastLowBar >= divRangeMin and candidateBar - lastLowBar <= divRangeMax and val > val[1]
// Linea hasta el extremo; etiqueta en la vela donde se detecto el giro.
if showDivs and earlyBearStart
    line.new(lastHighBar, lastHigh, candidateBar, val[1], color=color.orange, style=line.style_dashed)
    label.new(bar_index, val, "PRE Bear\nGiro 1 vela", style=label.style_label_down, color=color.orange, textcolor=color.white, size=size.tiny)
if showDivs and earlyBullStart
    line.new(lastLowBar, lastLow, candidateBar, val[1], color=color.aqua, style=line.style_dashed)
    label.new(bar_index, val, "PRE Bull\nGiro 1 vela", style=label.style_label_up, color=color.aqua, textcolor=color.black, size=size.tiny)

if barstate.isconfirmed and not na(momHigh)
    barsSinceHigh = (bar_index - pivotLen) - lastHighBar
    priceHigh     = high[pivotLen]

    if not na(lastHigh) and momHigh < lastHigh and priceHigh > lastHighPrice and barsSinceHigh >= divRangeMin and barsSinceHigh <= divRangeMax
        momBearDiv := true
        if showDivs and showConfirmed
            line.new(lastHighBar, lastHigh, bar_index - pivotLen, momHigh, width=2, color=color.red)
            label.new(bar_index - pivotLen, momHigh, "Bear Div", style=label.style_label_down, color=color.red, textcolor=color.white, size=size.tiny)

    lastHigh      := momHigh
    lastHighPrice := priceHigh
    lastHighBar   := bar_index - pivotLen

if barstate.isconfirmed and not na(momLow)
    barsSinceLow = (bar_index - pivotLen) - lastLowBar
    priceLow     = low[pivotLen]

    if not na(lastLow) and momLow > lastLow and priceLow < lastLowPrice and barsSinceLow >= divRangeMin and barsSinceLow <= divRangeMax
        momBullDiv := true
        if showDivs and showConfirmed
            line.new(lastLowBar, lastLow, bar_index - pivotLen, momLow, width=2, color=color.lime)
            label.new(bar_index - pivotLen, momLow, "Bull Div", style=label.style_label_up, color=color.lime, textcolor=color.white, size=size.tiny)

    lastLow      := momLow
    lastLowPrice := priceLow
    lastLowBar   := bar_index - pivotLen

//==============================================================================
// ALERTAS (STATIC & DYNAMIC)
//==============================================================================
// Todos los avisos se evalúan al cierre; un solo aviso por candidato.
alertRsi = input.bool(true, "Alertar RSI al cierre", group="Alertas")
alertEarly = input.bool(true, "Alertar preaviso con giro de 1 vela", group="Alertas")
alertConfirmed = input.bool(false, "Alertar tambien divergencias confirmadas (5 velas)", group="Alertas")
string signals = ""
if alertEarly and earlyBullStart
    signals += "PRE Bull: giro alcista confirmado por 1 vela. Preaviso, puede fallar.\n"
if alertEarly and earlyBearStart
    signals += "PRE Bear: giro bajista confirmado por 1 vela. Preaviso, puede fallar.\n"
if alertRsi and buySignal
    signals += "BUY RSI confirmado al cierre.\n"
if alertRsi and sellSignal
    signals += "SELL RSI confirmado al cierre.\n"
if alertConfirmed and momBullDiv
    signals += "Divergencia alcista CONFIRMADA.\n"
if alertConfirmed and momBearDiv
    signals += "Divergencia bajista CONFIRMADA.\n"
if barstate.isconfirmed and str.length(signals) > 0
    alert(syminfo.tickerid + " / " + timeframe.period + "\n" + signals, alert.freq_once_per_bar_close)

plotshape(showDivs and showConfirmed and momBullDiv, title="Confirmacion Bull", style=shape.triangleup, location=location.bottom, color=color.lime, size=size.tiny, text="CONF")
plotshape(showDivs and showConfirmed and momBearDiv, title="Confirmacion Bear", style=shape.triangledown, location=location.top, color=color.red, size=size.tiny, text="CONF")
alertcondition(alertEarly and earlyBullStart, title="PRE Bull giro 1 vela", message="Preaviso alcista: giro de momentum confirmado por una vela cerrada. Puede fallar.")
alertcondition(alertEarly and earlyBearStart, title="PRE Bear giro 1 vela", message="Preaviso bajista: giro de momentum confirmado por una vela cerrada. Puede fallar.")
alertcondition(alertRsi and buySignal, title="BUY RSI", message="BUY RSI confirmado al cierre")
alertcondition(alertRsi and sellSignal, title="SELL RSI", message="SELL RSI confirmado al cierre")
alertcondition(alertConfirmed and momBullDiv, title="Bull confirmada", message="Divergencia alcista confirmada")
alertcondition(alertConfirmed and momBearDiv, title="Bear confirmada", message="Divergencia bajista confirmada")
